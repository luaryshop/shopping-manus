    match /ira_negocios/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /ira_categorias/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /ira_banners/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /ira_novidades/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /ira_plantoes/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /ira_configuracoes/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /ira_contatos/{id} {
      // Visitantes podem registrar somente um evento de clique; só o administrador lê.
      allow create: if request.resource.data.keys().hasOnly(['negocioId','tipo','origem','criadoEm'])
                    && request.resource.data.tipo in ['whatsapp','ligar','instagram','facebook']
                    && request.resource.data.negocioId is string && request.resource.data.negocioId.size() < 80
                    && request.resource.data.origem is string && request.resource.data.origem.size() < 40
                    && request.resource.data.criadoEm == request.time;
      allow read: if isAdmin();
      allow update, delete: if false;
    }
    match /ira_cadastros/{id} {
      // O formulário público cria pedido pendente; o painel administrativo decide o que fazer.
      allow create: if request.resource.data.keys().hasOnly(['nome','categoria','whatsapp','plano','periodo','aceite','origem','bairro','endereco','horario','servicos','descricao','telefone','site','instagram','status','criadoEm'])
                    && request.resource.data.nome is string && request.resource.data.nome.size() > 1 && request.resource.data.nome.size() <= 80
                    && request.resource.data.categoria is string && request.resource.data.categoria.size() <= 40
                    && request.resource.data.whatsapp is string && request.resource.data.whatsapp.size() <= 25
                    && (!('plano' in request.resource.data) || request.resource.data.plano in ['gratis','presenca','destaque','premium'])
                    && (!('aceite' in request.resource.data) || request.resource.data.aceite == true)
                    && (!('origem' in request.resource.data) || (request.resource.data.origem is string && request.resource.data.origem.size() <= 40))
                    && (!('periodo' in request.resource.data) || request.resource.data.periodo in [1,3,6,12])
                    && (!('bairro' in request.resource.data) || (request.resource.data.bairro is string && request.resource.data.bairro.size() <= 60))
                    && (!('endereco' in request.resource.data) || (request.resource.data.endereco is string && request.resource.data.endereco.size() <= 120))
                    && (!('horario' in request.resource.data) || (request.resource.data.horario is string && request.resource.data.horario.size() <= 80))
                    && (!('servicos' in request.resource.data) || (request.resource.data.servicos is string && request.resource.data.servicos.size() <= 200))
                    && (!('descricao' in request.resource.data) || (request.resource.data.descricao is string && request.resource.data.descricao.size() <= 300))
                    && (!('telefone' in request.resource.data) || (request.resource.data.telefone is string && request.resource.data.telefone.size() <= 16))
                    && (!('site' in request.resource.data) || (request.resource.data.site is string && request.resource.data.site.size() <= 120))
                    && (!('instagram' in request.resource.data) || (request.resource.data.instagram is string && request.resource.data.instagram.size() <= 80))
                    && request.resource.data.status == 'novo'
                    && request.resource.data.criadoEm == request.time;
      allow read, update, delete: if isAdmin();
    }
    match /ira_mensagens/{id} {
      // Textos de venda/atendimento: só o administrador lê e edita.
      allow read, write: if isAdmin();
    }
